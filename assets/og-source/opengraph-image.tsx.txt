import { ImageResponse } from "next/og";
import { readFile } from "node:fs/promises";
import { join } from "node:path";
import { contourSvg, svgUri } from "@/lib/ogArt";

export const alt = "XERO — Season 2. Two friends. Too many thoughts. One microphone.";
export const dynamic = "force-static";
export const size = { width: 1200, height: 630 };
export const contentType = "image/png";

export default async function OpenGraphImage() {
  const [anton, mono] = await Promise.all([
    readFile(join(process.cwd(), "assets/fonts/Anton-Latin.ttf")),
    readFile(join(process.cwd(), "assets/fonts/JetBrainsMono-Latin.ttf")),
  ]);
  return new ImageResponse(
    (
      <div
        style={{
          width: "100%",
          height: "100%",
          display: "flex",
          flexDirection: "column",
          alignItems: "center",
          justifyContent: "center",
          background: "#000",
          position: "relative",
          color: "#f4f1ea",
        }}
      >
        {/* eslint-disable-next-line @next/next/no-img-element */}
        <img src={svgUri(contourSvg(1200, 630))} width={1200} height={630} alt="" style={{ position: "absolute", left: 0, top: 0 }} />
        <div style={{ display: "flex", fontFamily: "Mono", fontSize: 22, letterSpacing: 7, color: "rgba(244,241,234,0.75)" }}>
          SEASON 2 — A LIFE IN PROGRESS
        </div>
        <div
          style={{
            display: "flex",
            fontFamily: "Anton",
            fontSize: 260,
            lineHeight: 1,
            marginTop: 6,
            color: "#fff",
            textShadow: "6px 0 0 rgba(255,42,140,0.7), -6px 0 0 rgba(40,230,240,0.7)",
          }}
        >
          XERO
        </div>
        <div style={{ display: "flex", fontFamily: "Anton", fontSize: 40, letterSpacing: 1, marginTop: 10, textTransform: "uppercase" }}>
          Two friends. Too many thoughts. One microphone.
        </div>
        <div style={{ display: "flex", marginTop: 26, fontFamily: "Mono", fontSize: 18, letterSpacing: 5, color: "rgba(244,241,234,0.6)" }}>
          A HINGLISH PODCAST BY TWO BEST FRIENDS
        </div>
      </div>
    ),
    {
      ...size,
      fonts: [
        { name: "Anton", data: anton, style: "normal", weight: 400 },
        { name: "Mono", data: mono, style: "normal", weight: 400 },
      ],
    },
  );
}
